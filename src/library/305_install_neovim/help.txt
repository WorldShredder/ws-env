Options:
  --neovim-tag TAG  Git tag to clone from. Defaults to the latest version.
  --neovim-commit COMMIT
                    Git commit sha to checkout. Defaults to the latest version.
  --neovim-purge, --purge
                    Purge system of all known install locations before
                    performing install.
  --neovim-use-pkgman, --pkgman
                    Install using system package manager rather than building
                    from source.
